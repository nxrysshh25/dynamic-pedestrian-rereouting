import { useState, useEffect, useRef } from 'react'
import Sidebar from './components/Sidebar'
import MapCanvas from './components/MapCanvas'
import StatsOverlay from './components/StatsOverlay'
import SimulationControls from './components/SimulationControls'
import RouteExplanationPanel from './components/RouteExplanationPanel'
import './index.css'

function App() {
  const [routeData, setRouteData] = useState(null);
  const [isLoading, setIsLoading] = useState(false);
  const [error, setError] = useState(null);
  
  const [simState, setSimState] = useState(null);
  const [injectMode, setInjectMode] = useState(false);
  
  // Store last requested route parameters for recalculation
  const lastRequest = useRef(null);

  useEffect(() => {
    fetch('http://127.0.0.1:8000/api/simulation/state')
      .then(res => res.json())
      .then(setSimState)
      .catch(console.error);
  }, []);

  const handleRouteCalculation = async (origin, destination, profile) => {
    setIsLoading(true);
    setError(null);
    setRouteData(null);
    lastRequest.current = { origin, destination, profile };
    
    try {
      const response = await fetch('http://127.0.0.1:8000/api/route', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ origin, destination, profile }),
      });
      
      if (!response.ok) {
        throw new Error('Failed to calculate route. Please try different locations.');
      }
      
      const data = await response.json();
      setRouteData(data);
    } catch (err) {
      setError(err.message);
    } finally {
      setIsLoading(false);
    }
  };

  const handleRecalculate = () => {
    if (lastRequest.current) {
      const { origin, destination, profile } = lastRequest.current;
      handleRouteCalculation(origin, destination, profile);
    }
  };

  const handleInjectCongestion = async (lon, lat) => {
    if (!injectMode) return;
    try {
      await fetch('http://127.0.0.1:8000/api/simulation/inject_congestion', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ lon, lat, penalty: 200.0 }) // High penalty
      });
      setInjectMode(false);
      handleRecalculate();
    } catch (e) {
      console.error('Failed to inject congestion', e);
    }
  };

  return (
    <div style={{ display: 'flex', width: '100%', height: '100%' }}>
      <Sidebar 
        onCalculate={handleRouteCalculation} 
        isLoading={isLoading} 
        error={error} 
      />
      <div style={{ flex: 1, position: 'relative' }}>
        <MapCanvas 
          routeData={routeData} 
          injectMode={injectMode}
          onMapClick={handleInjectCongestion}
        />
        {routeData && <StatsOverlay stats={routeData.stats} profile={routeData.route.properties.profile} />}
        {routeData && <RouteExplanationPanel stats={routeData.stats} route={routeData.route} />}
        
        <SimulationControls 
          simState={simState} 
          setSimState={setSimState} 
          onRecalculate={handleRecalculate}
          onInjectCongestion={setInjectMode}
        />
      </div>
    </div>
  )
}

export default App

import React, { useEffect, useRef, useState } from 'react';
import Map, { Source, Layer } from 'react-map-gl/maplibre';
import * as maplibregl from 'maplibre-gl';
import 'maplibre-gl/dist/maplibre-gl.css';

export default function MapCanvas({ routeData, injectMode, onMapClick }) {
  const mapRef = useRef();
  
  // Default map view centered on Melbourne CBD
  const [viewState, setViewState] = useState({
    longitude: 144.965,
    latitude: -37.813,
    zoom: 14.5,
    pitch: 45,
    bearing: 0
  });

  const [sensors, setSensors] = useState([]);
  const [signals, setSignals] = useState([]);

  useEffect(() => {
    fetch('http://127.0.0.1:8000/api/sensors')
      .then(res => res.json())
      .then(setSensors)
      .catch(console.error);
      
    fetch('http://127.0.0.1:8000/api/signals')
      .then(res => res.json())
      .then(setSignals)
      .catch(console.error);
  }, []);

  useEffect(() => {
    if (routeData && mapRef.current) {
      const coords = routeData.route.geometry.coordinates;
      if (coords && coords.length > 0) {
        // Find bounding box to fit the route
        const bounds = coords.reduce(
          (acc, coord) => [
            [Math.min(acc[0][0], coord[0]), Math.min(acc[0][1], coord[1])],
            [Math.max(acc[1][0], coord[0]), Math.max(acc[1][1], coord[1])]
          ],
          [[coords[0][0], coords[0][1]], [coords[0][0], coords[0][1]]]
        );
        
        mapRef.current.fitBounds(bounds, { padding: 80, duration: 1500 });
      }
    }
  }, [routeData]);

  const handleMapClick = (e) => {
    if (injectMode && onMapClick) {
      onMapClick(e.lngLat.lng, e.lngLat.lat);
    }
  };

  // Route Style
  const routeLayerStyle = {
    id: 'route',
    type: 'line',
    layout: {
      'line-join': 'round',
      'line-cap': 'round'
    },
    paint: {
      'line-color': '#2563eb', // accent color
      'line-width': 6,
      'line-opacity': 0.8
    }
  };

  // Restricted segments style
  const restrictedLayerStyle = {
    id: 'restricted-segments',
    type: 'line',
    layout: {
      'line-join': 'round',
      'line-cap': 'round'
    },
    paint: {
      'line-color': '#dc2626', // danger color
      'line-width': 8,
      'line-opacity': 0.9,
      'line-dasharray': [2, 2]
    }
  };

  const cursorStyle = injectMode ? 'crosshair' : 'grab';

  return (
    <div style={{ position: 'absolute', top: 0, bottom: 0, left: 0, right: 0 }}>
      <Map
        ref={mapRef}
        {...viewState}
        onMove={evt => setViewState(evt.viewState)}
        onClick={handleMapClick}
        mapStyle="https://basemaps.cartocdn.com/gl/positron-gl-style/style.json"
        mapLibre={maplibregl}
        style={{ width: '100%', height: '100%' }}
        cursor={cursorStyle}
      >
        {/* Sensors Layer */}
        {sensors.length > 0 && (
          <Source id="sensors-source" type="geojson" data={{
            type: 'FeatureCollection',
            features: sensors.map(s => ({
              type: 'Feature',
              geometry: { type: 'Point', coordinates: [s.longitude, s.latitude] },
              properties: { id: s.location_id, name: s.sensor_name }
            }))
          }}>
            <Layer
              id="sensors-layer"
              type="circle"
              paint={{
                'circle-radius': 4,
                'circle-color': '#1890ff',
                'circle-stroke-width': 1,
                'circle-stroke-color': '#ffffff'
              }}
            />
          </Source>
        )}

        {/* Signals Layer */}
        {signals.length > 0 && (
          <Source id="signals-source" type="geojson" data={{
            type: 'FeatureCollection',
            features: signals.map(s => ({
              type: 'Feature',
              geometry: { type: 'Point', coordinates: [s.LONGITUDE, s.LATITUDE] },
              properties: { id: s.SITE_NO, name: s.SITE_NAME }
            }))
          }}>
            <Layer
              id="signals-layer"
              type="circle"
              paint={{
                'circle-radius': 3,
                'circle-color': '#faad14',
                'circle-stroke-width': 1,
                'circle-stroke-color': '#000000'
              }}
            />
          </Source>
        )}

        {routeData && (
          <>
            {/* Main Route */}
            <Source id="route-source" type="geojson" data={routeData.route}>
              <Layer {...routeLayerStyle} />
            </Source>

            {/* Restricted Segments */}
            {routeData.restricted_segments.features.length > 0 && (
              <Source id="restricted-source" type="geojson" data={routeData.restricted_segments}>
                <Layer {...restrictedLayerStyle} />
              </Source>
            )}

            {/* Origin Marker */}
            <Source id="origin-source" type="geojson" data={{
              type: 'Feature',
              geometry: {
                type: 'Point',
                coordinates: routeData.route.geometry.coordinates[0]
              }
            }}>
              <Layer
                id="origin-marker"
                type="circle"
                paint={{
                  'circle-radius': 8,
                  'circle-color': '#15803d',
                  'circle-stroke-width': 2,
                  'circle-stroke-color': '#ffffff'
                }}
              />
            </Source>

            {/* Destination Marker */}
            <Source id="dest-source" type="geojson" data={{
              type: 'Feature',
              geometry: {
                type: 'Point',
                coordinates: routeData.route.geometry.coordinates[routeData.route.geometry.coordinates.length - 1]
              }
            }}>
              <Layer
                id="dest-marker"
                type="circle"
                paint={{
                  'circle-radius': 8,
                  'circle-color': '#b91c1c',
                  'circle-stroke-width': 2,
                  'circle-stroke-color': '#ffffff'
                }}
              />
            </Source>
          </>
        )}
      </Map>
    </div>
  );
}

import React, { useState, useEffect } from 'react';

export default function SimulationControls({ simState, setSimState, onInjectCongestion, onRecalculate }) {
  const [isPlaying, setIsPlaying] = useState(false);
  const [injectMode, setInjectMode] = useState(false);

  useEffect(() => {
    let interval;
    if (isPlaying) {
      interval = setInterval(async () => {
        try {
          const res = await fetch('http://127.0.0.1:8000/api/simulation/step', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ delta_seconds: 300 }) // advance 5 mins
          });
          const data = await res.json();
          setSimState(data);
          onRecalculate();
        } catch (e) {
          console.error(e);
          setIsPlaying(false);
        }
      }, 2000);
    }
    return () => clearInterval(interval);
  }, [isPlaying, setSimState, onRecalculate]);

  const handleReset = async () => {
    const res = await fetch('http://127.0.0.1:8000/api/simulation/reset', { method: 'POST' });
    const data = await res.json();
    setSimState(data);
    onRecalculate();
  };

  const formatTime = (secs) => {
    const mins = Math.floor(secs / 60) % 60;
    const hrs = Math.floor(secs / 3600);
    return `+${hrs}h ${mins}m`;
  };

  return (
    <div style={{
      position: 'absolute',
      bottom: 20,
      left: 20,
      background: 'white',
      padding: '15px',
      borderRadius: '8px',
      boxShadow: '0 4px 6px rgba(0,0,0,0.1)',
      zIndex: 10,
      display: 'flex',
      flexDirection: 'column',
      gap: '10px',
      width: '320px'
    }}>
      <h3 style={{ margin: 0, fontSize: '16px' }}>Simulation Controls</h3>
      
      <div style={{ display: 'flex', justifyContent: 'space-between', fontSize: '14px' }}>
        <span><strong>Time:</strong> {simState?.date} {simState?.hour}:00</span>
        <span><strong>Elapsed:</strong> {formatTime(simState?.elapsed_seconds || 0)}</span>
      </div>

      <div style={{ display: 'flex', gap: '8px' }}>
        <button 
          onClick={() => setIsPlaying(!isPlaying)}
          style={{ flex: 1, padding: '8px', background: isPlaying ? '#ff4d4f' : '#1890ff', color: 'white', border: 'none', borderRadius: '4px', cursor: 'pointer' }}
        >
          {isPlaying ? 'Pause' : 'Play (x300)'}
        </button>
        <button 
          onClick={handleReset}
          style={{ flex: 1, padding: '8px', background: '#f0f0f0', border: '1px solid #d9d9d9', borderRadius: '4px', cursor: 'pointer' }}
        >
          Reset
        </button>
      </div>

      <button 
        onClick={() => {
          setInjectMode(!injectMode);
          onInjectCongestion(!injectMode); // toggle mode in parent
        }}
        style={{ padding: '8px', background: injectMode ? '#52c41a' : '#faad14', color: 'white', border: 'none', borderRadius: '4px', cursor: 'pointer', marginTop: '5px' }}
      >
        {injectMode ? 'Click Map to Inject Congestion...' : 'Inject High Congestion'}
      </button>
    </div>
  );
}
